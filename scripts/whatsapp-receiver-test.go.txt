package main

import (
    "os"
    "testing"
    "time"
    "go.mau.fi/whatsmeow/types"
    "go.mau.fi/whatsmeow/types/events"
    waProto "go.mau.fi/whatsmeow/proto/waE2E"
    waLog "go.mau.fi/whatsmeow/util/log"
    "google.golang.org/protobuf/proto"
)

func TestPastorOnlyArchive(t *testing.T) {
    old, _ := os.Getwd()
    if err := os.Chdir(t.TempDir()); err != nil { t.Fatal(err) }
    defer os.Chdir(old)
    pastorJID = "233200000001@s.whatsapp.net"
    store, err := NewMessageStore()
    if err != nil { t.Fatal(err) }
    defer store.Close()
    if err := store.StoreChat("233200000002@s.whatsapp.net", "Other", time.Now()); err == nil {
        t.Fatal("other chat was accepted")
    }
    pastor, _ := types.ParseJID(pastorJID)
    other, _ := types.ParseJID("233200000002@s.whatsapp.net")
    for _, test := range []struct{id string; sender types.JID; fromMe, group bool}{
        {"other-sender", other, false, false}, {"outgoing", pastor, true, false},
        {"group", pastor, false, true}, {"allowed", pastor, false, false},
    } {
        msg := &events.Message{Info: types.MessageInfo{MessageSource: types.MessageSource{
            Chat: pastor, Sender: test.sender, IsFromMe: test.fromMe, IsGroup: test.group,
        }, ID: test.id, Timestamp: time.Now()}, Message: &waProto.Message{Conversation: proto.String("Test message")}}
        handleMessage(nil, store, msg, waLog.Noop)
    }
    var count int
    if err := store.db.QueryRow("SELECT count(*) FROM messages").Scan(&count); err != nil { t.Fatal(err) }
    if count != 1 { t.Fatalf("expected only one allowed incoming message, got %d", count) }
    var id string
    store.db.QueryRow("SELECT id FROM messages").Scan(&id)
    if id != "allowed" { t.Fatal("wrong message stored") }
}
